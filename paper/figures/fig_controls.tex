% Fig. 2 -- the control ladder (TikZ, one column). Each arm adds ONE ingredient of the attack,
% so consecutive differences isolate one component of ASR. Requires tikz libraries: positioning,
% arrows.meta, decorations.pathreplacing, calc.
\begin{tikzpicture}[
    font=\footnotesize, >={Stealth[length=3.2pt]},
    arm/.style={draw=black!55, rounded corners=2pt, fill=white, minimum height=7.2mm, text width=45mm, align=left, inner xsep=3pt},
    tag/.style={font=\scriptsize, text=black!60},
    br/.style={decorate, decoration={brace, amplitude=3pt, raise=1pt}, draw=black!60, line width=0.5pt},
]
\node[arm] (a0) {\textbf{Clean}: no injection \hfill {\scriptsize ASR $\equiv 0$}};
\node[arm, below=2.2mm of a0] (a1) {\textbf{Placebo}: paraphrase \emph{every} item};
\node[arm, below=2.2mm of a1] (a2) {\textbf{Count-matched}: paraphrase only the items the attack injected};
\node[arm, below=2.2mm of a2] (a3) {\textbf{Bias-matched}: $+$ the attack's daily bias prompts};
\node[arm, draw=attack, fill=attack!7, line width=0.7pt, below=2.2mm of a3] (a4) {\textbf{Attack} (frozen tape): $+$ misinformation content};
\foreach \i/\j/\t in {a0/a1/{retrieval noise}, a1/a2/{article count}, a2/a3/{bias prompts}, a3/a4/{meaning}} {
  \draw[br] ([xshift=2mm]\i.east) -- ([xshift=2mm]\j.east) node[midway, right=5pt, tag, align=left] {\t};
}
\draw[->, black!45] ([xshift=-3mm]a0.west) -- ([xshift=-3mm]a4.west) node[midway, left, tag, rotate=90, anchor=south] {one ingredient added per step};
\end{tikzpicture}
